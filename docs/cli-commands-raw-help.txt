          Self-learning replay JSONL path [default: data/replay.jsonl]
      --self-learn-state-dir <SELF_LEARN_STATE_DIR>
          Self-learning state directory for adapters and metrics [default: adapters/selflearn]
      --forgetting-manifest <FORGETTING_MANIFEST>
          Capability probe manifest path (enables forgetting analysis)
      --forgetting-store <FORGETTING_STORE>
          Forgetting curves store path (defaults to <state-dir>/forgetting_curves.json)
      --forgetting-jsonl <FORGETTING_JSONL>
          Optional JSONL export path for forgetting deltas
      --forgetting-threshold <FORGETTING_THRESHOLD>
          Absolute capability-score delta treated as significant forgetting [default: 0.02]
      --forgetting-max-examples <FORGETTING_MAX_EXAMPLES>
          Maximum examples evaluated per capability probe [default: 8]
      --forgetting-allow-code-exec
          Allow code execution in capability probes
      --require-all-probes
          Require every manifest probe to run; otherwise missing probes are skipped
      --forgetting-baseline-id <FORGETTING_BASELINE_ID>
          Baseline checkpoint or session id used to compute forgetting deltas
  -h, --help
          Print help
```
